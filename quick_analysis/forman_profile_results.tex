\begin{frame}{Models}
\begin{itemize}
    \item \textbf{GeneralSheaf}: learns unconstrained restriction maps on each edge via a neural sheaf learner. The learned sheaf Laplacian is used for signal diffusion.
    \item \textbf{JointSheafParams (Joint Diffusion)}: jointly diffuses both node signals and restriction maps at each layer. Two initial-map configurations are considered:
    \begin{itemize}
        \item \textbf{Identity initial maps} (\texttt{learn\_first\_maps=False}): restriction maps are initialised to the identity and then updated by joint diffusion.
        \item \textbf{Learned general initial maps} (\texttt{learn\_first\_maps=True}, \texttt{learnt\_map\_type=general}): a neural learner provides a general initial map before joint diffusion proceeds.
    \end{itemize}
    \item \textbf{Normalisation configurations}: each model is run with \texttt{normalised=True} (symmetric normalisation $\widetilde{\Delta} = D^{-1/2}\Delta D^{-1/2}$) and \texttt{normalised=False} (no normalisation).
\end{itemize}
\end{frame}

\begin{frame}{Forman Profile and Alignment Metrics}
\begin{itemize}
    \item \textbf{Forman curvature profile:} at the last training epoch and for each fold, the diagonal block $F_{u,u}$ of the Forman curvature endomorphism is extracted from the learned sheaf Laplacian. Its $d$ eigenvalues (where $d$ is the stalk dimension) form the per-node \emph{Forman profile}.
    \item \textbf{Topological reference:} the Forman curvature of the augmented normalised graph Laplacian (fixed, topology-only quantity).
    \item \textbf{Alignment metrics} (computed for each eigenvalue channel $k=1,\dots,d$, then averaged over folds):
    \begin{itemize}
        \item Spearman $\rho$ and Pearson $r$: node-level rank/linear correlation with the topological profile.
        \item Wasserstein distance: between the z-scored empirical distributions.
        \item RMSE: $\ell_2$ between raw Forman values and topological values.
        \item NRMSE: $\ell_2$ between z-scored profiles.
    \end{itemize}
    \item \textbf{Heatmaps:} rows = training checkpoints (epoch-0 $\to$ last), columns = layers; each cell = metric value averaged over eigenvalue channels. Fixed colour scales: $[-1,1]$ for correlations (diverging), $[0, 0.9]$ for Wasserstein, $[0, 0.6]$ for RMSE, $[0, 1.8]$ for NRMSE.
\end{itemize}
\end{frame}

\begin{frame}{Cross-comparison Configurations}
In addition to comparing the Forman profile of a trained model to the topological reference under the \emph{same} normalisation, two cross-normalization comparisons are performed:
\begin{itemize}
    \item \textbf{Norm=True sheaf vs Norm=False topology:} the model is trained with \texttt{normalised=True}; its Forman profile is compared to the topological Forman curvature computed from the \emph{unnormalised} augmented Laplacian.
    \item \textbf{Norm=False sheaf vs Norm=True topology:} the model is trained with \texttt{normalised=False}; its Forman profile is compared to the topological curvature from the \emph{normalised} augmented Laplacian.
\end{itemize}
These cross-comparisons probe whether the learned Forman structure aligns with a topological reference that differs from the one implicitly used during training.
\end{frame}
